\begin{lemma}
	\label{lemma:hoeff}
	(Hoeffding's Lemma) Let $X$ be a scalar r.v taking values in
	$[a,b]$. Then for any $t > 0$
	\begin{equation}
		\label{lemma:hoef:eq1}
		\E e^{tX} \leq e^{t\Ex X}(1 + O(t^2\sigma^2\exp(O(t(b-a))))).
	\end{equation}
	In particular
	\begin{equation}
		\label{lemma:hoef:eq2}
		\E e^{tX} \leq e^{t\E X}\exp(O(t^2(b-a)^2)).
	\end{equation}
\end{lemma}

\begin{proof}
	Normalize $X$ as to $\Ex(X) = 0$ and $b - a = 1$. And consider the
	Taylor expansion of $e^{tX}$,
	\[
		e^{tX} = 1 + tX + O(t^2X^2\exp(O(t)))
	\]
	taking expectation yields the first result. The second one follows
	from assymptotics of the exponential function.
\end{proof}

\begin{theorem}
	\label{trm:McDiamird}
	(McDiamird's Theorem) Let $(R_i)_{i=1}^n$ be a tuple of ranges
	and $X = (X_i)_{i=1}^{n}$ a random vector taking values in
	their product.
	Let $F: \prod_{i=1}^n R_i \to \R$ be a coordinate-wise bounded
	function (this is a strong Lipchitz condition). That is changing
	any coordinate $x_i$ may change at most $c_i$ the total value.
	Then
	\[
		\Pr( |F(X) - \Ex F(X)| > \lambda \sigma) \leq C \exp(-c \lambda^2)
	\]
	for some $C,c > 0$ and $\sigma^2 = \sum_{i=1}^n c_i^2$.
\end{theorem}

\begin{proof}
	By simmetry, we may only consider $\Pr(F(X) - \E F(X) > \lambda
	\sigma)$. We turn to Hoeffding's Lemma [\ref{lemma:hoeff}],
	consider
	\[
		\Ex \exp(tF(X)) = \Ex \big[ \Ex[\exp(tF(X))| X_1, \dots X_{n-1}] \big],
	\]
	we should try to understand $\Ex[\exp(tF(X))| X_1, \dots X_{n-1}]$.
	We have that
	\[
		\Ex[\exp(tF(X))| X_1, \dots X_{n-1}] = \Ex[\exp(t(F(X) -
		\Ex_{X_n} F(X)))\exp(t\Ex F(X))| X_1, \dots X_{n-1}]
	\]
	applying (\ref{lemma:hoef:eq2}) to the first product, (here we are
	conditioning on $X_1, \dots X_{n-1}$)
	\[
		\Ex[\exp(t(F(X) - \Ex_{X_n}F(X)))\exp(t\Ex F(X))| - ] \leq
		\exp(O(t^2c_n^2))\Ex[\exp(tF(X)) | -]
	\]
	and we can iteratively apply the theorem to the new coordinate-wise
	bounded function
	\[
		G(X_1, \dots X_{n-1}) = \Ex[F(X)|X_1, \dots X_{n-1}].
	\]
	Notice $G$ has the same coefficients as $F$, because
	\begin{align*}
		|G(y, x_2 \dots, x_{n-1}) - G(z, x_2 \dots, x_{n-1})| &= \bigg|
		\int_{R_n} F(y,\dots,x_n) dx_n - \int_{R_n} F(z,\dots,x_n) dx_n\bigg|\\
		&= \bigg| \int_{R_n} F(y,\dots,x_n) - F(z,\dots,x_n) dx_n\bigg| < c_1,
	\end{align*}
	and here we can take any other coordinate instead of the first.
\end{proof}

\begin{corollary}
	\label{trm:hoeffding_ineq}
	(Hoeffding's inequality) Let $(X_i)_{i=1}^n$ be real r.v's taking
	value in bounded intervals $[a_i,b_i]$. Then, if $S = \sum_{i \leq
	n} X_i$, we have
	\[
		\Pr( |S  - \Ex S| > \lambda\sigma) < C\exp(-c \lambda^2)
	\]
	for some $C,c > 0$, where $\sigma^2 = \sum_{i \leq n} (b_i-a_i)^2$.
\end{corollary}
\begin{proof}
	Apply [\ref{trm:McDiamird}] with $F(X_1, \dots X_n) = X_1 + \dots + X_n$.
\end{proof}

\begin{theorem}
	\label{trm:chernoff} (Chernoff) Let $(X_i)_{i=1}^n$ be independent
	scalar random variables with $|X_i| < K$ a.s,
	with mean $\mu_i$, and variance $\sigma_i^2$. Then for any $\lambda
	> 0$, one has
	\[
		\Pr(|S_n - \mu| \geq \lambda \sigma) \leq C\exp(-c \min(\lambda^2,
		\frac{\lambda \sigma}{K}))
	\]
	for some absolute $C,c > 0$, where $S_n = \sum_{i=1}^{n} X_i$, $\mu
	= \sum_{i=1}^{n} \mu_i$ and $\sigma^2 = \sum_{i=1}^{n} \sigma_i^2$.
\end{theorem}

\begin{proof}
	The proof is again the use of the exponential moment with lemma
	[\ref{lemma:hoeff}]. Suppose we normalize  each variable
	so that $K \leq 1$ and $\mu_i = 0$, then $\mu = 0$ and for any $0
	\leq t \leq 1$,
	\[
		\Ex(\exp(tS_n)) = \prod_{i=1}^{n} \Ex(\exp(tX_i)) = \prod_{i=1}^n
		(1 + O(t^2\sigma_i^2\exp(O(t)))).
	\]
	using the usual $1 + x \leq e^x$ ineq, we find
	\[
		\Ex(\exp(tS_n)) \leq \exp(t\Ex(S_n))\exp(O(t^2)\exp(O(t))\sigma^2)
		= \exp(O(t^2\sigma^2)),
	\]
	now we use Markov and optimize on $t$ following the restraint on $0
	\leq t \leq 1$. (Suppose the hidden constant is $D$)
	we have that
	\[
		\Pr(S_n \geq \lambda\sigma) \leq \exp(D t^2\sigma^2 - t\lambda\sigma)
	\]
	which yields the result after optimizing $t$ following the
	constraints. The other inequality is equivalent.

\end{proof}

\begin{corollary}
	\label{lemma:expec_binomial_bounds}
	(Tailbounds for the Binomial distribution) Let $B(n,p)$ be the
	binomial distribution.
	We have that
	\[
		\Pr(|B(n,p) - np| \geq \delta np) \leq C\exp(-c \delta^2 np)
	\]
	for some absolute $C,c$.
\end{corollary}
\begin{proof}
	Note that
	\[
		\Pr(|B(n,p) - np| \geq \lambda \sqrt{np}) \leq \Pr(|B(n,p) - np|
		\geq \lambda \sqrt{np(1-p)}) \leq C\exp(-c\lambda^2)
	\]
	by \ref{trm:chernoff}. Applying it with $\lambda = \delta
	\sqrt{np}$ yields the result.
\end{proof}

\begin{theorem}
	\label{trm:azuma_inequality}
	(Azuma's Inequality) Let $X_1, X_2, \dots X_n$ be a sequence scalar
	r.v's with $|X_i| \leq 1$ almost surely. Assume also
	that we have the martingale difference property
	\[
		\E(X_i | X_1, \dots, X_{i-1}) = 0
	\]
	almost surely for all $i = 1,\dots,n$. Then for any $\lambda > 0$,
	the sum $S_n := X_1 + \dots + X_n$ satisfies
	\[
		\Pr(|S_n| \geq \lambda \sqrt{n}) \leq C\exp(-c\lambda^2)
	\]
	for some absolute $C,c > 0$.
\end{theorem}

\begin{proof}
	We prove this by calculating exponential moments and using
	Hoeffding's Lemma again. Note that $S_n = S_{n-1} + X_n$ so that
	\[
		\Ex(e^{tS_n}) = \Ex(e^{tS_{n-1}}e^{tX_n}).
	\]
	By taking the expectation of conditioning on the first $n-1$
	values, we get that
	\[
		\Ex(e^{tS_n}) = \Ex \Ex (e^{tS_{n-1}}e^{X_n} | X_1, \dots,
		X_{n-1}) =\Ex [e^{tS_{n-1}}\Ex(e^{tX_n} | X_1, \dots, X_{n-1})]
	\]
	Now, by \ref{lemma:hoeff}, we have that $\Ex(e^{tX_n} | X_1, \dots,
	X_{n-1}) \leq e^{O(t^2)}$, so by iterating we get the result.
\end{proof}

\begin{theorem} [Subexponential Bernstein inequality]
	\label{trm:bernstein_ineq}
	Let $X_1, \dots, X_n$ be independent, mean $0$, subexponential r.v's.
 Then, for every $t \geq 0$, we have
	\[
		\Pr\bigg\{\bigg|\sum_{i=1}^n X_i\bigg| \geq t\bigg\} \leq
		2\exp\bigg(-c\min\bigg(\frac{t^2}{\sum_{j = 1}^{n}
		||X_j||_{\psi_1}^2},\frac{t}{\max_j {||X_j}||_{\psi_1}}\bigg)\bigg)
	\]
	where $c$ is an absolute constant.
\end{theorem}
\begin{proof}
	The proof is again only using the exponential Markov method, but as
	in the proof of Chernoff, we have some limiting on $\lambda$ as the
	variables are only subexponential.
\end{proof}

\begin{theorem}[Janson Inequalities]
	\label{trm:janson_ineq}
	Let $A_1, A_2 \dots A_t \subset [N]$ be a family of sets. Consider the events
	$B_j = \{A_j \subset [N]_p\}$. Then
	\begin{equation}
		\Pr\bigg( \bigcap_{j=1}^t B_j^c\bigg) \leq
		\begin{cases}
			\exp(-\mu + \Delta/2)\\
			\exp(-\mu^2/2\Delta) & \text{ if } \mu \leq \Delta
		\end{cases}
	\end{equation}
	where
	\[\mu = \sum_{i =1 }^{t} \Pr(B_j)\]
	and
	\[
		\Delta = \sum_{i \sim j}\Pr(B_i \cap B_j)
	\]
	(here the sum being over ordered pairs).
\end{theorem}

\begin{remark}
	We also have the trivial lowerbound
	\[
		\Pr(\bigcap_{j=1}^{t} B_j^c) \geq \prod_{j = 1}^t \Pr(B_j^c).
	\]
\end{remark}

\begin{remark}
    The following is a version of the previous inequality with more general 
    lowerbounds (and it has a better presentation as well).
\end{remark}
\begin{theorem}
    Let $t_1, \dots, t_n$ be boolean random variables. Let $\mathcal{A}$ be a 
    collection of sets $A \subset [n]$ (we may think of it as hypergraph edges).
    Consider the random variable (or the random boolean polynomial)
    \[
    X = \sum_{A \in \mathcal{A}} \prod_{j \in A} t_j.
    \]
    We have 
    \[
    \Pr(X \leq \Ex X - T) \leq \exp(-T^2/\Delta)
    \]
    where $\Delta$ is defined just as before being 
    \[
        \Delta = \sum_{A \cap B \neq \varnothing} \Ex \prod_{j \in A \cup B} t_j.
    \]
\end{theorem}

\begin{proof}[Tao's proof of \ref{trm:janson_ineq}]
    Let's see a beautiful proof different from the one 
    in the Probabilistic Method. We will rely as before on the exponential method.
    Note that, defining $F(t) = \Ex \exp(-tX)$ and using Markov, we get that 
    \[
    \Pr(X \leq \Ex X - T) = \Pr(\exp(-tX) \geq \exp(-t (\Ex X - T)) 
    \leq F(t) \exp(-t(\Ex X - T)).
    \]
    So, taking logs, an upper bound on $\log F(t) -t(\Ex X - T)$ yields 
    an upper bound on our probability. Janson's ingenious trick is noting 
    that $F(0) = 1$ and writing the $\log F(t)$ as an integral, so that
    \[
    \log F(t) -t(\Ex X - T) = \int_0^t \frac{F'(s)}{F(s)} ds -t(\Ex X - T).
    \]
    Now, $F'(t) = \Ex[-X \exp(-tX)]$ so the last expression becomes 
    \[
        - \int_0^t \frac{\Ex[X \exp(-sX)]}{F(s)} ds - t(\Ex X - T).
    \]
    So flipping signs, we want a lower bound on 
    \[
        \int_0^t \frac{\Ex[X \exp(-sX)]}{F(s)} ds + t(\Ex X - T)
    \]
    Now we remind that $X$ is of the form $\sum_{A \in \mathcal{A}} \prod_{j \in A} t_j$
    and we use linearity inside the the integral to get
    \[
        \sum_{A \in \mathcal{A}}\int_0^t \frac{\Ex[(\prod_{j \in A} t_j)\exp(-yX)]}{\Ex \exp(-yX)} dy
    \]
    Letting $E_A$ be the event that for all $j \in A$, $t_j = 1$, we find that 
    \[
        \Ex[(\prod_{j \in A} t_j) \exp(-yX)] = \Pr(E_A)\Ex[\exp(-yX) | E_A].
    \]
    Writing $X = Y_A + Z_A$ where 
    \[
    Y_A = \sum_{\substack{B \in \mathcal{A}\\ A \cap B \neq \varnothing}} \prod_{j \in B} t_j
    \quad \text{and} \quad 
    Z_A = \sum_{\substack{B \in \mathcal{A}\\ A \cap B = \varnothing}} \prod_{j \in B} t_j
    \]
    we get that 
    \[
        \Ex[\exp(-yX) | E_A] = \Ex[\exp(-y(Y_A + Z_A)) | E_A] = 
        \Ex[\exp(-yZ_A)] \Ex[\exp(-y Y_A) | E_A].
    \]
    substituting back in the sum, we get 
    \[
        \sum_{A \in \mathcal{A}} \Pr(E_A) \int_0^t \frac{\Ex[\exp(-sZ_A)]\Ex[\exp(-s Y_A)|E_A]}{F(s)} ds.
    \]
    Since clearly $Z_A \leq X$, we have that $\Ex[\exp(-sZ_A)] \geq F(s)$, from which 
    the independent term cancels. So we get a lowerbound 
    \[
    \sum_{A \in \mathcal{A}} \Pr(E_A) \int_0^t \Ex[\exp(-s Y_A)|E_A] ds
    \]
    Applying Jensen's inequality for $\Ex[\exp(-s Y_A)|E_A] \geq \exp(-s \Ex[Y_A | E_A])$
    we get a new lowerbound of 
    \[
        \int_0^t \sum_{A \in \mathcal{A}} \Pr(E_A) \exp(-s \Ex[Y_A | E_A]) ds.
    \]
    We may apply Jensen's again using the fact that $\sum_{A \in \mathcal{A}} \Pr(E_A) = \Ex X$
    to get that this last integral is lower bounded by 
    \[
        \Ex X \int_0^t \exp\bigg( -s\sum_{A \in \mathcal{A}} \frac{\Pr(E_A)\Ex[Y_A|E_A]}{\Ex X}\bigg) ds = \Ex X \int_0^t \exp(-s \frac{\Delta}{\Ex X}) ds  
    \]
    we thus conclude by noticing that we can lowerbound the full expression by 
    \[
        \frac{(\Ex X)^2}{\Delta} (1 - \exp(-t \frac{\Delta}{\Ex X})) -t(\Ex X - T).
    \]
    maximizing over $t$ (or setting it $T/\Delta$) we get the result.
\end{proof}
